if(NOT BOOP_BUILD_PROGRAMS)
  return()
endif()

file(GLOB BOOP_HELO_BENCHMARKS CONFIGURE_DEPENDS benchmarks/*.aig)
foreach(BOOP_HELO_BENCHMARK ${BOOP_HELO_BENCHMARKS})
  get_filename_component(BOOP_HELO_TEST_NAME ${BOOP_HELO_BENCHMARK} NAME_WE)
  set(BOOP_HELO_TEST_ARGS
    -DTEST_NAME=helo_${BOOP_HELO_TEST_NAME}
    -DHELO_EXECUTABLE=$<TARGET_FILE:helo>
    -DINPUT_PATH=${BOOP_HELO_BENCHMARK}
    -DOUTPUT_PATH=${CMAKE_CURRENT_BINARY_DIR}/${BOOP_HELO_TEST_NAME}.out.aig
  )

  if(BOOP_USE_ABC)
    list(APPEND BOOP_HELO_TEST_ARGS
      -DABC_EXECUTABLE=$<TARGET_FILE:abc>
    )
  endif()

  add_test(NAME helo_${BOOP_HELO_TEST_NAME}
    COMMAND ${CMAKE_COMMAND}
      ${BOOP_HELO_TEST_ARGS}
      -P ${CMAKE_CURRENT_SOURCE_DIR}/run_helo.cmake
  )
endforeach()
